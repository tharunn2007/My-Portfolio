   131→      if (menuRef.current && !menuRef.current.contains(e.target)) {
   132→        onClose();

   135→

   137→      if (e.key === "Escape") {
   138→        onClose();

   140→        (e.key === "Backspace" || e.key === "Delete") &&
   141→        (e.metaKey || e.ctrlKey || menu?.type === "folder" || menu?.type === "shortcut")
   142→      ) {
   143→        if (menu?.type === "folder" && menu?.target) {
   144→          e.preventDefault();

   146→            onDeleteFolder(menu.target);
   147→          } else if (onDeleteCustomFolder) {
   148→            onDeleteCustomFolder(menu.target?.id);
   149→          }
   150→          onClose();

   154→            onDeleteShortcut(menu.target.id);
   155→          }
   156→          onClose();

   162→    window.addEventListener("keydown", handleKeyDown);
   163→    window.addEventListener("resize", onClose);

   166→      window.removeEventListener("pointerdown", handlePointerDown);
   167→      window.removeEventListener("keydown", handleKeyDown);
   168→      window.removeEventListener("resize", onClose);
